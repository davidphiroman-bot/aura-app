btnGal.className = "flex items-center gap-2 px-3 py-1.5 rounded-lg text-xs font-medium transition-all bg-cyan-500 text-black shadow"; btnSingle.className = "flex items-center gap-2 px-3 py-1.5 rounded-lg text-xs font-medium text-gray-400 hover:text-white transition-all"; } else { galSection.classList.add('hidden'); singleSection.classList.remove('hidden');

// --- ANIMACIÓN DE CANVAS EN VIVO --- function startCanvasAnimations() { // Main Hub Canvas const hubCanvas = document.getElementById('hub-main-canvas'); if (hubCanvas) { const ctx = hubCanvas.getContext('2d'); let step = 0;

function drawHub() { hubCanvas.width = hubCanvas.clientWidth; hubCanvas.height = hubCanvas.clientHeight; ctx.clearRect(0, 0, hubCanvas.width, hubCanvas.height);

ctx.beginPath(); ctx.strokeStyle = '#06b6d4'; ctx.lineWidth = 2;

for (let x = 0; x < hubCanvas.width; x += 10) { const y = hubCanvas.height / 2 + Math.sin((x + step) * 0.05) * 25 + Math.cos(x * 0.02) * 10; if (x === 0) ctx.moveTo(x, y); else ctx.lineTo(x, y); } ctx.stroke();

step += 2; requestAnimationFrame(drawHub); } drawHub(); }
